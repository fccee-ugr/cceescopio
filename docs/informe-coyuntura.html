Plantilla de informe de coyuntura

Esqueleto, guía de redacción y rúbrica para que el estudiantado elabore su propio informe con datos reales y actualizados

Fecha de modificación

24 de noviembre de 2026

Un informe de coyuntura describe y explica la situación económica reciente de un territorio: cómo crece la actividad, qué ocurre con el empleo y los precios, cómo están las cuentas públicas y el sector exterior, y qué riesgos hay. Es el tipo de documento que publican cada trimestre el Banco de España, los servicios de estudios y las cámaras de comercio, y un buen ejercicio para aprender a leer, relacionar e interpretar indicadores.

Esta página ofrece todo lo necesario para hacer uno en clase: una plantilla generada con los últimos datos del territorio que elijas, la estructura del informe, una guía de redacción y la rúbrica de evaluación. Como modelo, puedes consultar el Boletín de coyuntura de CCEEscopio.

function jsonstat(js) {
  const ids = js.id, size = js.size, dims = js.dimension;
  const codes = ids.map(d => {
    const idx = dims[d].category.index;
    return Array.isArray(idx) ? idx : Object.keys(idx).sort((a, b) => idx[a] - idx[b]);
  });
  const labels = ids.map(d => dims[d].category.label || {});
  const vals = js.value, out = [];
  const keys = Array.isArray(vals) ? vals.map((v, i) => i).filter(i => vals[i] != null) : Object.keys(vals).map(Number);
  for (const i of keys) {
    let rem = i; const rec = {};
    for (let d = ids.length - 1; d >= 0; d--) {
      const c = codes[d][rem % size[d]];
      rec[ids[d]] = c;
      if (ids[d] === "geo") rec.geoLabel = labels[d][c] || c;
      rem = Math.floor(rem / size[d]);
    }
    rec.value = +vals[i];
    out.push(rec);
  }
  return out;
}

preferidas = ["TOTAL", "T", "TOT", "CP00", "PC", "EUR", "NR", "SA", "SCA", "RCH_A", "I90", "A", "Y15-74"]
function unaCategoria(recs) {
  // Para cada dimensión distinta de geo y time, conserva una sola categoría
  if (!recs.length) return recs;
  const dimsAll = Object.keys(recs[0]).filter(k => !["geo", "time", "value", "geoLabel"].includes(k));
  let r = recs;
  for (const d of dimsAll) {
    const cats = [...new Set(r.map(x => x[d]))];
    if (cats.length < 2) continue;
    const elegida = preferidas.find(p => cats.includes(p)) ?? cats[0];
    r = r.filter(x => x[d] === elegida);
  }
  return r;
}

async function eurostat(code, query = "") {
  const url = `https://ec.europa.eu/eurostat/api/dissemination/statistics/1.0/data/${code}?format=JSON&lang=EN${query}`;
  const res = await fetch(url);
  if (!res.ok) {
    let msg = `HTTP ${res.status}`;
    try { const js = await res.json(); const err = Array.isArray(js.error) ? js.error[0] : js.error; if (err && err.label) msg += ` – ${err.label}`; } catch (e) {}
    throw new Error(`Eurostat ${code}: ${msg}`);
  }
  return jsonstat(await res.json());
}

function fecha(t) {
  let m;
  if ((m = /^(\d{4})-(\d{2})$/.exec(t))) return new Date(+m[1], +m[2] - 1, 1);
  if ((m = /^(\d{4})M(\d{2})$/.exec(t))) return new Date(+m[1], +m[2] - 1, 1);
  if ((m = /^(\d{4})-?Q(\d)$/.exec(t))) return new Date(+m[1], (+m[2] - 1) * 3, 1);
  if ((m = /^(\d{4})-?S(\d)$/.exec(t))) return new Date(+m[1], (+m[2] - 1) * 6, 1);
  if ((m = /^(\d{4})$/.exec(t))) return new Date(+m[1], 0, 1);
  return null;
}

async function consultar(code, query, geos, desde) {
  try {
    return await eurostat(code, query + (geos ? geos.map(g => `&geo=${g}`).join("") : "") + desde);
  } catch (e1) {
    // Si algún territorio no existe en la tabla, se repite la consulta sin filtrar territorios
    const r = await eurostat(code, query + desde);
    return geos ? r.filter(x => geos.includes(x.geo)) : r;
  }
}

async function cargar(spec) {
  const desde = `&sinceTimePeriod=${spec.desde ?? 2008}`;
  const geos = spec.regional ? geosRegiones : (spec.sinGeo ? null : geosPaises);
  // Intentos: la consulta principal, las alternativas (p. ej., tablas que Eurostat ha renovado)
  // y, por último, la tabla sin filtros adicionales
  const intentos = [{code: spec.code, query: spec.query}, ...(spec.alt ?? []), {code: spec.code, query: ""}];
  let recs = null, usado = null, error = "";
  for (const t of intentos) {
    try { recs = await consultar(t.code, t.query, geos, desde); if (recs.length) { usado = t; break; } }
    catch (e) { error = e.message; }
  }
  if (!recs || !recs.length) return {ok: false, error: error || "la consulta no devolvió datos", rows: []};
  recs = unaCategoria(recs);
  const rows = recs.map(r => ({
    indicador: spec.nombre,
    codigo_territorio: r.geo ?? "",
    territorio: r.geo ? (nombresGeo[r.geo] ?? r.geoLabel) : (spec.serie ?? spec.nombre),
    periodo: r.time,
    fecha: fecha(r.time),
    valor: r.value,
    unidad: spec.unidad,
    fuente: `Eurostat (${usado.code})`
  })).filter(r => r.fecha && Number.isFinite(r.valor));
  return {ok: rows.length > 0, rows, code: usado.code, error: rows.length ? "" : "la consulta no devolvió datos"};
}

function rewind(f) {
  // Los GeoJSON de GISCO siguen la regla de la mano derecha; d3 necesita el sentido contrario
  const fix = poly => d3.geoArea({type: "Polygon", coordinates: poly}) > 2 * Math.PI ? poly.map(r => r.slice().reverse()) : poly;
  const g = f.geometry;
  if (g?.type === "Polygon") g.coordinates = fix(g.coordinates);
  else if (g?.type === "MultiPolygon") g.coordinates = g.coordinates.map(fix);
  return f;
}

async function geoNuts(nivel) {
  // Límites de países (nivel 0) o regiones NUTS 2 (nivel 2) de Eurostat-GISCO; los países tienen
  // como alternativa Natural Earth. Solo se descargan cuando se pide un mapa y se guardan en memoria.
  window.__cceeGeo = window.__cceeGeo ?? new Map();
  if (!window.__cceeGeo.has(nivel)) window.__cceeGeo.set(nivel, geoDescargar(nivel));
  return window.__cceeGeo.get(nivel);
}

async function geoDescargar(nivel) {
  const url = `https://gisco-services.ec.europa.eu/distribution/v2/nuts/geojson/NUTS_RG_20M_2021_4326_LEVL_${nivel}.geojson`;
  try {
    const r = await fetch(url);
    if (r.ok) { const g = await r.json(); g.features.forEach(f => { rewind(f); f.properties.geo = f.properties.NUTS_ID; }); return g; }
  } catch (e) {}
  if (nivel === 0) {
    try {
      const r = await fetch("https://cdn.jsdelivr.net/gh/nvkelso/natural-earth-vector@master/geojson/ne_110m_admin_0_countries.geojson");
      const g = await r.json(), eq = {GR: "EL", GB: "UK"};
      g.features.forEach(f => { const c = f.properties.ISO_A2_EH ?? f.properties.ISO_A2; f.properties.geo = eq[c] ?? c; });
      return g;
    } catch (e) {}
  }
  return null;
}

function tooltipMapa(node) {
  // Etiqueta propia para los mapas: aparece en cuanto el ratón está sobre cualquier territorio
  // (la de Plot solo se activa cerca del centro del territorio y fallaba en los grandes).
  const tip = window.__cceeTip ?? (window.__cceeTip = Object.assign(document.createElement("div"), {className: "mapa-tip", hidden: true}));
  if (!tip.isConnected) document.body.appendChild(tip);
  // Los contornos destacados y los nombres no deben tapar a los territorios
  node.querySelectorAll('svg g[fill="none"], svg g[aria-label="text"]').forEach(g => { g.style.pointerEvents = "none"; });
  node.querySelectorAll("svg title").forEach(t => {
    const el = t.parentNode; el.dataset.tip = t.textContent; el.classList.add("mapa-zona"); t.remove();
  });
  let actual = null;
  const ocultar = () => { tip.hidden = true; if (actual) actual.classList.remove("mapa-hl"); actual = null; };
  const mover = e => {
    const el = e.target.closest ? e.target.closest("[data-tip]") : null;
    if (!el || !node.contains(el)) return ocultar();
    if (el !== actual) {
      if (actual) actual.classList.remove("mapa-hl");
      actual = el; el.classList.add("mapa-hl");
      const [l1, ...resto] = el.dataset.tip.split("\n");
      tip.innerHTML = `<b>${l1}</b>${resto.length ? "<br>" + resto.join("<br>") : ""}`;
      tip.hidden = false;
    }
    const x = e.clientX + 14, y = e.clientY + 14, w = tip.offsetWidth, h = tip.offsetHeight;
    tip.style.left = Math.min(x, window.innerWidth - w - 8) + "px";
    tip.style.top = (y + h > window.innerHeight - 8 ? e.clientY - h - 10 : y) + "px";
  };
  node.addEventListener("pointermove", mover);
  node.addEventListener("pointerdown", mover);
  node.addEventListener("pointerleave", ocultar);
  window.addEventListener("scroll", ocultar, {passive: true});
  return node;
}

Estructura del informe

El informe tiene ocho secciones. La extensión orientativa es de 1.500 a 2.500 palabras con cuatro a seis gráficos o tablas.

Sección Qué responder Indicadores de CCEEscopio
1. Resumen Tres a cinco frases con los mensajes principales: cómo crece la economía, qué pasa con el empleo y los precios, y cuál es el principal riesgo. Escríbelo al final. Síntesis propia
2. Actividad y demanda ¿Crece la economía? ¿Más o menos que hace un año y que la UE? ¿Qué componentes de la demanda tiran del crecimiento? ¿Qué anticipan los indicadores de confianza? PIB: crecimiento interanual (trimestral); Consumo de los hogares: crecimiento interanual; Inversión (formación bruta de capital fijo): crecimiento interanual; Índice de producción industrial; Indicador de sentimiento económico
3. Mercado laboral ¿Baja el paro? ¿A quién afecta más? ¿Se crea empleo? Compara con la UE y con el año anterior, en puntos porcentuales. Tasa de paro; Tasa de paro juvenil (menores de 25 años); Tasa de empleo (20-64 años)
4. Precios ¿Sube la inflación o se modera? ¿Es un fenómeno de energía y alimentos o también de la inflación subyacente? ¿Qué ocurre con la vivienda? IPCA: inflación general (tasa anual); IPCA: inflación subyacente (sin energía ni alimentos); Precio de la vivienda (tasa anual)
5. Sector público y condiciones financieras ¿Cómo están el déficit y la deuda respecto a las reglas fiscales europeas (3 % y 60 % del PIB)? ¿Cómo evolucionan los tipos de interés y qué efecto tienen en hogares y empresas? Déficit (−) o superávit (+) público (% del PIB); Deuda pública (% del PIB); Rentabilidad del bono a 10 años; Euríbor a 3 meses
6. Sector exterior ¿Cómo evolucionan las exportaciones? ¿La economía tiene capacidad o necesidad de financiación frente al resto del mundo? Exportaciones: crecimiento interanual (trimestral); Saldo por cuenta corriente (% del PIB)
7. Riesgos y perspectivas Señala dos o tres riesgos concretos (al alza o a la baja) y justifícalos con datos. ¿Qué indicadores habrá que vigilar en los próximos meses? Síntesis propia
8. Conclusión Valoración global en un párrafo: ¿la situación mejora, empeora o se estanca? No repitas el resumen: interpreta. Síntesis propia

Guía de redacción

  • Dato, comparación e interpretación. Cada afirmación necesita las tres cosas. Por ejemplo (cifras ilustrativas): «La tasa de paro bajó al 10,4 % en septiembre (dato), 0,8 puntos menos que hace un año y aún el doble que en la UE (comparación), gracias al empleo en los servicios (interpretación)».
  • Fecha y unidad siempre. Indica el periodo de cada dato (mes, trimestre o año) y su unidad. Los indicadores se publican con retrasos distintos: no mezcles datos de periodos diferentes sin avisarlo.
  • Puntos frente a porcentajes. Si el paro pasa del 11 % al 10 %, baja 1 punto porcentual, no un 1 %: baja un 9,1 %. Para variables en porcentaje, usa los puntos.
  • Tasa frente a nivel. Que la inflación baje del 3 % al 2 % no significa que los precios bajen: suben más despacio. Revisa Detecta el engaño.
  • Interanual frente a intertrimestral. Las tasas interanuales comparan con el mismo periodo del año anterior y evitan la estacionalidad; las intertrimestrales detectan antes los cambios de tendencia.
  • Relaciona los indicadores. El informe gana valor cuando conecta secciones: consumo y confianza, empleo y producción, tipos de interés y vivienda, inflación y salarios.
  • Gráficos honestos y citados. Usa el generador de gráficos, que añade la fuente, y comenta cada gráfico en el texto.
  • Escribe el resumen al final, cuando ya sepas cuáles son los mensajes principales.

Rúbrica de evaluación

Cada criterio se valora con uno de los cuatro niveles; la nota del criterio es su peso multiplicado por el porcentaje del nivel. Por ejemplo, «Adecuado» en un criterio de 2,5 puntos suma 1,75.

Criterio Puntos Excelente (100 %) Adecuado (70 %) Mejorable (40 %) Insuficiente (0 %)
Uso correcto de los datos 2,5 Cifras exactas, con periodo y unidad; distingue tasas, niveles y puntos porcentuales. Datos correctos con algún descuido de periodo o unidad. Errores en algunos datos o confunde tasas y puntos. Datos erróneos, sin fecha o sin unidad.
Interpretación económica 2,5 Explica causas y relaciones entre indicadores; compara con la UE y con el pasado. Interpreta los datos, pero con pocas relaciones o comparaciones. Describe los datos sin apenas interpretarlos. Se limita a copiar cifras.
Estructura y síntesis 1,5 Resumen con los mensajes clave, secciones equilibradas y extensión ajustada. Estructura clara, con un resumen mejorable. Estructura desordenada o resumen ausente. No sigue la estructura del informe.
Gráficos y tablas 1,5 Pertinentes, con título, unidad y fuente, y comentados en el texto. Correctos, pero alguno sin fuente o sin comentar. Poco pertinentes o con errores de presentación. Sin gráficos o gráficos engañosos.
Riesgos y perspectivas 1,0 Riesgos concretos, justificados con datos. Riesgos razonables, poco justificados. Riesgos genéricos. No identifica riesgos.
Redacción y fuentes 1,0 Clara, sin errores; cita las fuentes y la fecha de consulta. Algún error menor de redacción o de cita. Errores frecuentes o fuentes incompletas. Redacción deficiente y sin fuentes.

Antes de entregar

Para el profesorado

  • Un territorio por grupo. Asigna a cada grupo un país de la UE o una comunidad autónoma (con el comparador de comunidades autónomas) y dedica una sesión a comparar los diagnósticos.
  • Evaluación entre iguales. Cada grupo evalúa el informe de otro con la rúbrica; la tabla de autoevaluación ya viene en la plantilla.
  • Seguimiento a lo largo del curso. Repite el informe al principio y al final del cuatrimestre con el mismo territorio: los datos de Eurostat se actualizan solos y el estudiantado ve cómo cambia la coyuntura.
  • Encaje en las asignaturas. Funciona en Macroeconomía, Economía Española y Mundial, Coyuntura Económica, Estructura Económica y en las asignaturas de Métodos Cuantitativos (números índice, tasas de variación, series temporales).
