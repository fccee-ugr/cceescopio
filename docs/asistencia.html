dniValido = (x) => {
  const s = String(x || "").toUpperCase().replace(/[^0-9A-Z]/g, ""), L = "TRWAGMYFPDXBNJZSQVHLCKE";
  let m = s.match(/^(\d{8})([A-Z])$/); if (m) return L[+m[1] % 23] === m[2] ? s : null;
  m = s.match(/^([XYZ])(\d{7})([A-Z])$/); if (m) return L[+("XYZ".indexOf(m[1]) + m[2]) % 23] === m[3] ? s : null;
  return null;
}
