Tamaño muestral

¿Cuántas observaciones necesita tu estudio? Elige el escenario y calcula n con los criterios clásicos de la literatura

Fecha de modificación

25 de noviembre de 2026

Antes de recoger datos (una encuesta, un experimento, un estudio de mercado, un TFG o un TFM), conviene decidir cuántas observaciones hacen falta. La respuesta depende del objetivo del estudio:

  • Estimar una proporción o una media con una precisión dada (error de muestreo máximo y nivel de confianza).
  • Contrastar una hipótesis con una potencia dada (probabilidad de detectar un efecto de un tamaño determinado si existe).

Elige el escenario que se ajusta a tu estudio, introduce los valores y la calculadora te da n, las fórmulas paso a paso y cómo cambia n si modificas los supuestos. Escribe los decimales con coma o con punto.

EST = {
  // Funciones estadísticas de CCEEscopio: distribuciones, regresión MCO y contrastes de supuestos.
  // Lo incrusta tools/make_apps4.py en las páginas de calculadoras; también se puede probar con Node.
  function crearEstadistica() {
    // ---------- distribuciones ----------
    const lgamma = (x) => {
      const c = [76.18009172947146, -86.50532032941677, 24.01409824083091, -1.231739572450155, 0.1208650973866179e-2, -0.5395239384953e-5];
      let y = x, tmp = x + 5.5; tmp -= (x + 0.5) * Math.log(tmp);
      let ser = 1.000000000190015; for (const ci of c) ser += ci / ++y;
      return -tmp + Math.log(2.5066282746310005 * ser / x);
    };
    const betacf = (a, b, x) => {
      const MAXIT = 300, EPS = 3e-16, FPMIN = 1e-300;
      let qab = a + b, qap = a + 1, qam = a - 1, c = 1, d = 1 - qab * x / qap;
      if (Math.abs(d) < FPMIN) d = FPMIN; d = 1 / d; let h = d;
      for (let m = 1; m <= MAXIT; m++) {
        const m2 = 2 * m;
        let aa = m * (b - m) * x / ((qam + m2) * (a + m2));
        d = 1 + aa * d; if (Math.abs(d) < FPMIN) d = FPMIN; c = 1 + aa / c; if (Math.abs(c) < FPMIN) c = FPMIN; d = 1 / d; h *= d * c;
        aa = -(a + m) * (qab + m) * x / ((a + m2) * (qap + m2));
        d = 1 + aa * d; if (Math.abs(d) < FPMIN) d = FPMIN; c = 1 + aa / c; if (Math.abs(c) < FPMIN) c = FPMIN; d = 1 / d;
        const del = d * c; h *= del; if (Math.abs(del - 1) < EPS) break;
      }
      return h;
    };
    // Beta incompleta regularizada I_x(a, b)
    const ibeta = (x, a, b) => {
      if (x <= 0) return 0; if (x >= 1) return 1;
      const bt = Math.exp(lgamma(a + b) - lgamma(a) - lgamma(b) + a * Math.log(x) + b * Math.log(1 - x));
      return x < (a + 1) / (a + b + 2) ? bt * betacf(a, b, x) / a : 1 - bt * betacf(b, a, 1 - x) / b;
    };
    // Gamma incompleta regularizada inferior P(a, x)
    const gammainc = (a, x) => {
      if (x <= 0) return 0;
      if (x < a + 1) { let ap = a, sum = 1 / a, del = sum; for (let n = 0; n < 500; n++) { ap++; del *= x / ap; sum += del; if (Math.abs(del) < Math.abs(sum) * 1e-15) break; } return sum * Math.exp(-x + a * Math.log(x) - lgamma(a)); }
      let b = x + 1 - a, c = 1e300, d = 1 / b, h = d;
      for (let i = 1; i < 500; i++) { const an = -i * (i - a); b += 2; d = an * d + b; if (Math.abs(d) < 1e-300) d = 1e-300; c = b + an / c; if (Math.abs(c) < 1e-300) c = 1e-300; d = 1 / d; const del = d * c; h *= del; if (Math.abs(del - 1) < 1e-15) break; }
      return 1 - Math.exp(-x + a * Math.log(x) - lgamma(a)) * h;
    };
    const pT = (t, df) => ibeta(df / (df + t * t), df / 2, 0.5);           // p-valor bilateral de la t de Student
    const pF = (f, d1, d2) => f <= 0 ? 1 : ibeta(d2 / (d2 + d1 * f), d2 / 2, d1 / 2); // p-valor de la F de Snedecor
    const pChi = (x, k) => x <= 0 ? 1 : 1 - gammainc(k / 2, x / 2);       // p-valor de la chi-cuadrado
    const normCdf = (z) => { const t = 1 / (1 + 0.2316419 * Math.abs(z)), d = 0.3989422804014327 * Math.exp(-z * z / 2);
      const p = d * t * (0.319381530 + t * (-0.356563782 + t * (1.781477937 + t * (-1.821255978 + t * 1.330274429)))); return z > 0 ? 1 - p : p; };
    const normInv = (p) => { // Acklam
      const a = [-39.69683028665376, 220.9460984245205, -275.9285104469687, 138.3577518672690, -30.66479806614716, 2.506628277459239];
      const b = [-54.47609879822406, 161.5858368580409, -155.6989798598866, 66.80131188771972, -13.28068155288572];
      const c = [-7.784894002430293e-3, -0.3223964580411365, -2.400758277161838, -2.549732539343734, 4.374664141464968, 2.938163982698783];
      const d = [7.784695709041462e-3, 0.3224671290700398, 2.445134137142996, 3.754408661907416];
      const q = Math.min(p, 1 - p);
      let x;
      if (q < 0.02425) { const r = Math.sqrt(-2 * Math.log(q)); x = (((((c[0] * r + c[1]) * r + c[2]) * r + c[3]) * r + c[4]) * r + c[5]) / ((((d[0] * r + d[1]) * r + d[2]) * r + d[3]) * r + 1); x = p < 0.5 ? x : -x; }
      else { const r = (p - 0.5), s = r * r; x = (((((a[0] * s + a[1]) * s + a[2]) * s + a[3]) * s + a[4]) * s + a[5]) * r / (((((b[0] * s + b[1]) * s + b[2]) * s + b[3]) * s + b[4]) * s + 1); }
      return x;
    };
    const tInv = (p, df) => { // cuantil bilateral: t tal que P(|T| > t) = p
      let lo = 0, hi = 1000; for (let i = 0; i < 200; i++) { const m = (lo + hi) / 2; if (pT(m, df) > p) lo = m; else hi = m; } return (lo + hi) / 2;
    };
  
    // ---------- álgebra ----------
    const invertir = (A) => {
      const n = A.length, M = A.map((r, i) => [...r, ...Array.from({length: n}, (_, j) => +(i === j))]);
      for (let c = 0; c < n; c++) {
        let p = c; for (let r = c + 1; r < n; r++) if (Math.abs(M[r][c]) > Math.abs(M[p][c])) p = r;
        if (Math.abs(M[p][c]) < 1e-12) return null;
        [M[c], M[p]] = [M[p], M[c]];
        const piv = M[c][c]; for (let j = 0; j < 2 * n; j++) M[c][j] /= piv;
        for (let r = 0; r < n; r++) if (r !== c) { const f = M[r][c]; if (f) for (let j = 0; j < 2 * n; j++) M[r][j] -= f * M[c][j]; }
      }
      return M.map(r => r.slice(n));
    };
    const media = (a) => a.reduce((s, x) => s + x, 0) / a.length;
  
    // ---------- regresión MCO ----------
    // y: vector; X: matriz n×k SIN columna de unos (se añade la constante)
    function mco(y, X, conConstante = true) {
      const n = y.length, Z = X.map(r => conConstante ? [1, ...r] : r.slice()), k = Z[0].length;
      const ZtZ = Array.from({length: k}, (_, i) => Array.from({length: k}, (_, j) => Z.reduce((s, r) => s + r[i] * r[j], 0)));
      const inv = invertir(ZtZ); if (!inv) return null;
      const Zty = Array.from({length: k}, (_, i) => Z.reduce((s, r, t) => s + r[i] * y[t], 0));
      const b = inv.map(r => r.reduce((s, v, j) => s + v * Zty[j], 0));
      const ajuste = Z.map(r => r.reduce((s, v, j) => s + v * b[j], 0)), e = y.map((v, t) => v - ajuste[t]);
      const scr = e.reduce((s, v) => s + v * v, 0), ym = media(y), sct = y.reduce((s, v) => s + (v - ym) ** 2, 0);
      const gl = n - k, s2 = scr / gl, se = inv.map((r, i) => Math.sqrt(s2 * r[i]));
      const r2 = conConstante ? 1 - scr / sct : 1 - scr / y.reduce((s, v) => s + v * v, 0);
      const r2a = 1 - (1 - r2) * (n - 1) / gl, q = k - (conConstante ? 1 : 0);
      const F = q > 0 ? (r2 / q) / ((1 - r2) / gl) : NaN;
      const ll = -n / 2 * (Math.log(2 * Math.PI) + Math.log(scr / n) + 1);
      return {n, k, b, se, t: b.map((v, i) => v / se[i]), p: b.map((v, i) => pT(v / se[i], gl)), ajuste, e, scr, sct, s2, gl, r2, r2a, F, pFv: q > 0 ? pF(F, q, gl) : NaN,
        aic: -2 * ll + 2 * k, bic: -2 * ll + k * Math.log(n), inv, Z};
    }
  
    // ---------- contrastes de supuestos ----------
    function supuestos(y, X, m) {
      const n = m.n, e = m.e, out = {};
      // Normalidad: Jarque-Bera
      const em = media(e), m2 = media(e.map(v => (v - em) ** 2)), m3 = media(e.map(v => (v - em) ** 3)), m4 = media(e.map(v => (v - em) ** 4));
      const S = m3 / m2 ** 1.5, K = m4 / m2 ** 2, JB = n / 6 * (S * S + (K - 3) ** 2 / 4);
      out.normalidad = {nombre: "Normalidad de los residuos (Jarque-Bera)", est: JB, p: pChi(JB, 2), asimetria: S, curtosis: K};
      // Homocedasticidad: Breusch-Pagan (versión de Koenker, n·R²)
      const e2 = e.map(v => v * v), bp = mco(e2, X);
      if (bp) { const LM = n * bp.r2; out.homocedasticidad = {nombre: "Homocedasticidad (Breusch-Pagan)", est: LM, p: pChi(LM, X[0].length)}; }
      // Autocorrelación: Durbin-Watson y Breusch-Godfrey de orden 1
      let num = 0; for (let t = 1; t < n; t++) num += (e[t] - e[t - 1]) ** 2;
      const DW = num / m.scr;
      const Xbg = X.map((r, t) => [...r, t ? e[t - 1] : 0]), bg = mco(e, Xbg);
      out.autocorrelacion = {nombre: "Ausencia de autocorrelación (Breusch-Godfrey, orden 1)", est: bg ? n * bg.r2 : NaN, p: bg ? pChi(n * bg.r2, 1) : NaN, dw: DW};
      // Forma funcional: RESET de Ramsey (ŷ² y ŷ³)
      const yh = m.ajuste, sd = Math.sqrt(media(yh.map(v => (v - media(yh)) ** 2))) || 1, c = media(yh);
      const Xr = X.map((r, t) => [...r, ((yh[t] - c) / sd) ** 2, ((yh[t] - c) / sd) ** 3]), rr = mco(y, Xr);
      if (rr && rr.gl > 0) { const Fr = ((m.scr - rr.scr) / 2) / (rr.scr / rr.gl); out.linealidad = {nombre: "Forma funcional lineal (RESET de Ramsey)", est: Fr, p: pF(Fr, 2, rr.gl)}; }
      // Multicolinealidad: factores de inflación de la varianza
      if (X[0].length >= 2) out.vif = X[0].map((_, j) => { const xj = X.map(r => r[j]), otras = X.map(r => r.filter((_, i) => i !== j)), a = mco(xj, otras); return a ? 1 / (1 - a.r2) : NaN; });
      return out;
    }
  
    // ---------- estadística descriptiva ----------
    function cuantil(ordenados, q) { const h = (ordenados.length - 1) * q, l = Math.floor(h); return ordenados[l] + (h - l) * ((ordenados[l + 1] ?? ordenados[l]) - ordenados[l]); }
    function describir(v) {
      const a = v.filter(Number.isFinite), n = a.length; if (!n) return null;
      const o = a.slice().sort((x, y) => x - y), m = media(a);
      const s2 = n > 1 ? a.reduce((s, x) => s + (x - m) ** 2, 0) / (n - 1) : NaN, s = Math.sqrt(s2);
      const m2 = a.reduce((s, x) => s + (x - m) ** 2, 0) / n, m3 = a.reduce((s, x) => s + (x - m) ** 3, 0) / n, m4 = a.reduce((s, x) => s + (x - m) ** 4, 0) / n;
      return {n, media: m, mediana: cuantil(o, 0.5), sd: s, var: s2, cv: m !== 0 ? 100 * s / Math.abs(m) : NaN, min: o[0], max: o[n - 1], q1: cuantil(o, 0.25), q3: cuantil(o, 0.75),
        iqr: cuantil(o, 0.75) - cuantil(o, 0.25), asimetria: m3 / m2 ** 1.5, curtosis: m4 / m2 ** 2 - 3, suma: a.reduce((s, x) => s + x, 0)};
    }
    function correlacion(a, b) { const p = a.map((x, i) => [x, b[i]]).filter(([x, y]) => Number.isFinite(x) && Number.isFinite(y)); if (p.length < 3) return {r: NaN, n: p.length, p: NaN};
      const mx = media(p.map(q => q[0])), my = media(p.map(q => q[1]));
      const r = p.reduce((s, [x, y]) => s + (x - mx) * (y - my), 0) / Math.sqrt(p.reduce((s, [x]) => s + (x - mx) ** 2, 0) * p.reduce((s, [, y]) => s + (y - my) ** 2, 0));
      const t = r * Math.sqrt((p.length - 2) / (1 - r * r)); return {r, n: p.length, p: pT(t, p.length - 2)}; }
  
    return {lgamma, ibeta, gammainc, pT, pF, pChi, normCdf, normInv, tInv, invertir, mco, supuestos, describir, cuantil, correlacion, media};
  }
  return crearEstadistica();
}
M = {
  // Cálculo del tamaño muestral para CCEEscopio (tamano-muestral.qmd).
  // Usa las distribuciones de tools/estadistica.js (EST). Lo incrusta tools/make_muestra.py; también se puede probar con Node:
  //   node tools/muestra.js
  function crearMuestra(EST) {
    const z = (p) => EST.normInv(p);
    const zAlfa = (alfa, colas) => z(1 - alfa / colas);
    const sube = (x) => Math.ceil(x - 1e-9);
    // Corrección por población finita (Cochran, 1977): n = n0 / (1 + (n0 - 1) / N)
    const finita = (n0, N) => (N && N > 0) ? n0 / (1 + (n0 - 1) / N) : n0;
  
    // ---------- distribuciones no centrales ----------
    const pois = (j, m) => Math.exp(-m + j * Math.log(m) - EST.lgamma(j + 1));
    const sumaPois = (m, f) => { // suma de Poisson(m) · f(j) alrededor de la moda
      if (m < 1e-12) return f(0);
      const j0 = Math.floor(m); let s = 0;
      for (let j = j0; j < j0 + 5000; j++) { const w = pois(j, m), t = w * f(j); s += t; if (w < 1e-14 && j > m + 10) break; }
      for (let j = j0 - 1; j >= 0; j--) { const w = pois(j, m); s += w * f(j); if (w < 1e-14) break; }
      return s;
    };
    const cdfF = (x, d1, d2) => x <= 0 ? 0 : 1 - EST.pF(x, d1, d2);
    const cdfFnc = (x, d1, d2, lam) => x <= 0 ? 0 : sumaPois(lam / 2, j => EST.ibeta(d1 * x / (d1 * x + d2), d1 / 2 + j, d2 / 2));
    const cdfChi = (x, k) => x <= 0 ? 0 : EST.gammainc(k / 2, x / 2);
    const cdfChinc = (x, k, lam) => x <= 0 ? 0 : sumaPois(lam / 2, j => EST.gammainc(k / 2 + j, x / 2));
    const cuantil = (cdf, p) => { let lo = 0, hi = 1; while (cdf(hi) < p && hi < 1e7) hi *= 2; for (let i = 0; i < 200; i++) { const m = (lo + hi) / 2; if (cdf(m) < p) lo = m; else hi = m; } return (lo + hi) / 2; };
    // Menor n entero (≥ nMin) con potencia(n) ≥ objetivo
    const menorN = (pot, objetivo, nMin) => {
      let lo = nMin, hi = nMin; while (pot(hi) < objetivo) { lo = hi; hi = hi * 2; if (hi > 1e7) return NaN; }
      while (hi - lo > 1) { const m = Math.floor((lo + hi) / 2); if (pot(m) >= objetivo) hi = m; else lo = m; }
      return pot(lo) >= objetivo ? lo : hi;
    };
  
    // ---------- escenarios ----------
    // Cada función devuelve {n0 (sin redondear), n (por grupo si procede), grupos, pasos: [[etiqueta, valor]], potencia?}
    const E = {};
    // 1. Proporción con error de muestreo absoluto (Cochran, 1977)
    E.propAbs = ({conf, p, e, N}) => { const za = zAlfa(1 - conf, 2), n0 = za * za * p * (1 - p) / (e * e), n1 = finita(n0, N);
      return {n0, nf: n1, n: sube(n1), grupos: 1, pasos: [["z", za], ["n₀ = z²·p·(1 − p) / e²", n0], ...(N ? [["n = n₀ / (1 + (n₀ − 1)/N)", n1]] : [])]}; };
    // 2. Proporción con error relativo: e = er · p
    E.propRel = ({conf, p, er, N}) => { const r = E.propAbs({conf, p, e: er * p, N}); r.pasos.unshift(["Error absoluto e = er · p", er * p]); return r; };
    // 3. Media con error absoluto, desviación típica σ
    E.mediaAbs = ({conf, sigma, e, N}) => { const za = zAlfa(1 - conf, 2), n0 = za * za * sigma * sigma / (e * e), n1 = finita(n0, N);
      return {n0, nf: n1, n: sube(n1), grupos: 1, pasos: [["z", za], ["n₀ = z²·σ² / e²", n0], ...(N ? [["n = n₀ / (1 + (n₀ − 1)/N)", n1]] : [])]}; };
    // 4. Media con error relativo: coeficiente de variación CV y error relativo er
    E.mediaRel = ({conf, cv, er, N}) => { const za = zAlfa(1 - conf, 2), n0 = za * za * cv * cv / (er * er), n1 = finita(n0, N);
      return {n0, nf: n1, n: sube(n1), grupos: 1, pasos: [["z", za], ["n₀ = z²·CV² / er²", n0], ...(N ? [["n = n₀ / (1 + (n₀ − 1)/N)", n1]] : [])]}; };
    // 5. Contraste sobre una proporción (aproximación normal)
    E.test1p = ({alfa, colas, pot, p0, p1}) => { const za = zAlfa(alfa, colas), zb = z(pot), num = za * Math.sqrt(p0 * (1 - p0)) + zb * Math.sqrt(p1 * (1 - p1)), n0 = num * num / ((p1 - p0) ** 2);
      return {n0, nf: n0, n: sube(n0), grupos: 1, pasos: [["z₁₋α" + (colas === 2 ? "/2" : ""), za], ["z₁₋β", zb], ["n = [z₁₋α·√(p₀q₀) + z₁₋β·√(p₁q₁)]² / (p₁ − p₀)²", n0]]}; };
    // 6. Contraste sobre una media: tamaño del efecto d = (μ₁ − μ₀)/σ, con corrección iterativa por la t de Student
    E.test1m = ({alfa, colas, pot, d}) => { const za = zAlfa(alfa, colas), zb = z(pot), nz = ((za + zb) / d) ** 2;
      const potT = (n) => { const gl = n - 1, tc = EST.tInv(alfa * (2 / colas), gl), del = d * Math.sqrt(n); return 1 - cdfT(tc, gl, del) + (colas === 2 ? cdfT(-tc, gl, del) : 0); };
      const n = menorN(potT, pot, 2);
      return {n0: nz, nf: n, n, grupos: 1, potencia: potT(n), pasos: [["z₁₋α" + (colas === 2 ? "/2" : ""), za], ["z₁₋β", zb], ["Aproximación normal: n = (z₁₋α + z₁₋β)² / d²", nz], ["Con la t de Student (t no central)", n]]}; };
    // 7. Comparación de dos proporciones (Fleiss, Levin y Paik, 2003), con corrección de continuidad opcional
    E.test2p = ({alfa, colas, pot, p1, p2, cc, k = 1}) => { const za = zAlfa(alfa, colas), zb = z(pot), pm = (p1 + k * p2) / (1 + k), d = Math.abs(p1 - p2);
      const num = za * Math.sqrt((1 + 1 / k) * pm * (1 - pm)) + zb * Math.sqrt(p1 * (1 - p1) + p2 * (1 - p2) / k), n0 = num * num / (d * d);
      const ncc = cc ? n0 / 4 * (1 + Math.sqrt(1 + 2 * (1 + 1 / k) / (n0 * d))) ** 2 : n0;
      return {n0, nf: ncc, n: sube(ncc), n2: sube(k * ncc), grupos: 2, pasos: [["z₁₋α" + (colas === 2 ? "/2" : ""), za], ["z₁₋β", zb], ["p̄ (proporción media)", pm], ["n₁ sin corrección", n0], ...(cc ? [["n₁ con corrección de continuidad", ncc]] : [])]}; };
    // 8. Comparación de dos medias (grupos independientes), d de Cohen, con la t de Student
    E.test2m = ({alfa, colas, pot, d, k = 1}) => { const za = zAlfa(alfa, colas), zb = z(pot), nz = (1 + 1 / k) * ((za + zb) / d) ** 2;
      const potT = (n) => { const n2 = Math.ceil(k * n), gl = n + n2 - 2, tc = EST.tInv(alfa * (2 / colas), gl), del = d / Math.sqrt(1 / n + 1 / n2); return 1 - cdfT(tc, gl, del) + (colas === 2 ? cdfT(-tc, gl, del) : 0); };
      const n = menorN(potT, pot, 2);
      return {n0: nz, nf: n, n, n2: Math.ceil(k * n), grupos: 2, potencia: potT(n), pasos: [["z₁₋α" + (colas === 2 ? "/2" : ""), za], ["z₁₋β", zb], ["Aproximación normal: n₁ = (1 + 1/k)(z₁₋α + z₁₋β)² / d²", nz], ["Con la t de Student (t no central)", n]]}; };
    // 9. Coeficiente de correlación de Pearson (transformación z de Fisher)
    E.correl = ({alfa, colas, pot, r}) => { const za = zAlfa(alfa, colas), zb = z(pot), C = 0.5 * Math.log((1 + r) / (1 - r)), n0 = ((za + zb) / C) ** 2 + 3;
      return {n0, nf: n0, n: sube(n0), grupos: 1, pasos: [["z₁₋α" + (colas === 2 ? "/2" : ""), za], ["z₁₋β", zb], ["C = ½·ln[(1 + r)/(1 − r)]", C], ["n = [(z₁₋α + z₁₋β)/C]² + 3", n0]]}; };
    // 10. Regresión lineal múltiple: contraste F de significación conjunta, f² de Cohen (λ = f²·n)
    E.regresion = ({alfa, pot, f2, k}) => { const potF = (n) => { const d2 = n - k - 1; if (d2 < 1) return 0; const fc = cuantil(x => cdfF(x, k, d2), 1 - alfa); return 1 - cdfFnc(fc, k, d2, f2 * n); };
      const n = menorN(potF, pot, k + 2);
      return {n0: n, nf: n, n, grupos: 1, potencia: potF(n), pasos: [["λ = f²·n (parámetro de no centralidad)", f2 * n], ["Grados de libertad", `${k} y ${n - k - 1}`], ["Regla de Green (1991), contraste global: 50 + 8k", 50 + 8 * k], ["Regla de Green (1991), coeficientes individuales: 104 + k", 104 + k]]}; };
    // 11. Contraste chi-cuadrado (bondad del ajuste o independencia), w de Cohen (λ = w²·n)
    E.chi = ({alfa, pot, w, gl}) => { const cc = cuantil(x => cdfChi(x, gl), 1 - alfa), potC = (n) => 1 - cdfChinc(cc, gl, w * w * n);
      const n = menorN(potC, pot, 2);
      return {n0: n, nf: n, n, grupos: 1, potencia: potC(n), pasos: [["Valor crítico χ²", cc], ["λ = w²·n (parámetro de no centralidad)", w * w * n]]}; };
  
    // t no central: P(T ≤ x) con gl grados de libertad y no centralidad del (integración numérica sobre la chi)
    function cdfT(x, gl, del) {
      // T = (Z + del) / sqrt(V/gl): P(T ≤ x) = E_V[Φ(x·sqrt(V/gl) − del)]; se integra en V con Simpson en la escala de sqrt(V)
      const a = 0, b = Math.sqrt(gl) + 12, m = 400, h = (b - a) / m; let s = 0;
      const dens = (u) => { if (u <= 0) return 0; const v = u * u; return 2 * u * Math.exp((gl / 2 - 1) * Math.log(v) - v / 2 - (gl / 2) * Math.log(2) - EST.lgamma(gl / 2)); };
      for (let i = 0; i <= m; i++) { const u = a + i * h, w = i === 0 || i === m ? 1 : i % 2 ? 4 : 2; s += w * dens(u) * EST.normCdf(x * u / Math.sqrt(gl) - del); }
      return s * h / 3;
    }
  
    // Ajustes finales: efecto de diseño (Kish, 1965) y tasa de respuesta esperada
    const ajustar = (n, deff = 1, resp = 1) => { const nd = n * deff, nr = nd / resp; return {nd, nr, n: sube(nr)}; };
    // Afijación proporcional por estratos: n_h = n · N_h / N (redondeo por el método del mayor resto)
    const afijacion = (n, estratos) => { const N = estratos.reduce((a, e) => a + e.N, 0); const q = estratos.map(e => n * e.N / N); const b = q.map(Math.floor);
      let resto = n - b.reduce((a, x) => a + x, 0); q.map((x, i) => [x - b[i], i]).sort((u, v) => v[0] - u[0]).forEach(([, i]) => { if (resto > 0) { b[i]++; resto--; } });
      return estratos.map((e, i) => ({...e, n: b[i], peso: e.N / N})); };
    return {E, finita, ajustar, afijacion, cdfT, cdfFnc, cdfChinc, zAlfa};
  }
  return crearMuestra(EST);
}

  const curvaP = est ? [] : [.5, .6, .7, .75, .8, .85, .9, .95, .99].map(p => { try { const r = M.E[esc]({...v, pot: p}); return {p, n: M.ajustar(r.nf, deff, resp).n}; } catch (e) { return null; } }).filter(d => d && Number.isFinite(d.n));
  const g1 = Plot.plot({height: 300, marginLeft: 60, x: {label: etq, grid: true}, y: {label: "n", grid: true, type: "log"},
    marks: [Plot.line(curva, {x: "x", y: "n", stroke: "#A6195E", strokeWidth: 2.5}), Plot.dot([{x: x0, n: calc.n1}], {x: "x", y: "n", r: 6, fill: "#F29200"}),
      Plot.tip(curva, Plot.pointer({x: "x", y: "n", title: d => `${etq}: ${f2(d.x)}\nn = ${fN(d.n)}`}))]});
  const g2 = est ? null : Plot.plot({height: 300, marginLeft: 60, x: {label: "Potencia (1 − β)", grid: true, tickFormat: d => locES.format(".0%")(d)}, y: {label: "n", grid: true},
    marks: [Plot.line(curvaP, {x: "p", y: "n", stroke: "#2C6FBB", strokeWidth: 2.5}), Plot.dot(curvaP, {x: "p", y: "n", fill: "#2C6FBB", r: 3}), Plot.dot([{p: v.pot, n: calc.n1}], {x: "p", y: "n", r: 6, fill: "#F29200"}),
      Plot.tip(curvaP, Plot.pointer({x: "p", y: "n", title: d => `Potencia ${fP(d.p)}: n = ${fN(d.n)}`}))]});
  return html`<div class="graf-2"><div><h4>n según ${etq.toLowerCase()} (escala logarítmica)</h4>${g1}</div>${g2 ? html`<div><h4>n según la potencia</h4>${g2}</div>` : ""}</div>
  <p class="nota">El punto naranja es tu escenario. ${est ? "Reducir el error a la mitad multiplica n aproximadamente por cuatro." : "Detectar efectos más pequeños o exigir más potencia aumenta mucho el tamaño muestral."}</p>`;
}

Criterios y valores de referencia

Escenario Fórmula o criterio Referencia
Proporción, error absoluto n_0 = z^2_{1-\alpha/2}\,p(1-p)/e^2; con p = 0{,}5 se obtiene el tamaño más conservador Cochran (1977)
Media, error absoluto n_0 = z^2_{1-\alpha/2}\,\sigma^2/e^2 Cochran (1977)
Error relativo n_0 = z^2_{1-\alpha/2}\,CV^2/e_r^2; en proporciones, CV^2 = (1-p)/p Cochran (1977); Lohr (2022)
Población finita n = n_0 / \left(1 + (n_0 - 1)/N\right) Cochran (1977)
Efecto de diseño n_{\text{diseño}} = n \cdot \text{deff} Kish (1965)
Una proporción n = \left[z_{1-\alpha}\sqrt{p_0q_0} + z_{1-\beta}\sqrt{p_1q_1}\right]^2/(p_1 - p_0)^2 Fleiss, Levin y Paik (2003)
Dos proporciones Fórmula de Fleiss, con corrección de continuidad opcional Fleiss, Levin y Paik (2003); Fleiss, Tytun y Ury (1980)
Una o dos medias t no central; tamaños del efecto d: 0,2 pequeño, 0,5 mediano, 0,8 grande Cohen (1988)
Correlación n = \left[(z_{1-\alpha} + z_{1-\beta})/C\right]^2 + 3, con C = \tfrac12\ln\frac{1+r}{1-r}; r: 0,1, 0,3, 0,5 Cohen (1988)
Regresión múltiple F no central con \lambda = f^2 n; f²: 0,02, 0,15, 0,35. Reglas prácticas: n \ge 50 + 8k y n \ge 104 + k Cohen (1988); Green (1991)
Chi-cuadrado Chi-cuadrado no central con \lambda = w^2 n; w: 0,1, 0,3, 0,5 Cohen (1988)

Consejos. Si no conoces la proporción, usa p = 0,5 (el caso más desfavorable). Para la desviación típica o el coeficiente de variación, usa un estudio piloto, un estudio previo o datos de CCEEscopio. En los contrastes, fija el tamaño del efecto mínimo que tenga interés práctico, no el que esperas encontrar. Si prevés que parte de la muestra no responderá, ajusta por la tasa de respuesta. Los resultados de las funciones de potencia coinciden con los de G*Power (Faul et al., 2007).
