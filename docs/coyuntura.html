Para la inflación se considera mejor estar cerca del 2 %, el objetivo del Banco Central Europeo. La columna Puntuación resume la posición media en todos los indicadores.

function jsonstat(js) {
  const ids = js.id, size = js.size, dims = js.dimension;
  const codes = ids.map(d => {
    const idx = dims[d].category.index;
    return Array.isArray(idx) ? idx : Object.keys(idx).sort((a, b) => idx[a] - idx[b]);
  });
  const labels = ids.map(d => dims[d].category.label || {});
  const vals = js.value, out = [];
  const keys = Array.isArray(vals) ? vals.map((v, i) => i).filter(i => vals[i] != null) : Object.keys(vals).map(Number);
  for (const i of keys) {
    let rem = i; const rec = {};
    for (let d = ids.length - 1; d >= 0; d--) {
      const c = codes[d][rem % size[d]];
      rec[ids[d]] = c;
      if (ids[d] === "geo") rec.geoLabel = labels[d][c] || c;
      rem = Math.floor(rem / size[d]);
    }
    rec.value = +vals[i];
    out.push(rec);
  }
  return out;
}

preferidas = ["TOTAL", "T", "TOT", "CP00", "PC", "EUR", "NR", "SA", "SCA", "RCH_A", "I90", "A", "Y15-74"]
function unaCategoria(recs) {
  // Para cada dimensión distinta de geo y time, conserva una sola categoría
  if (!recs.length) return recs;
  const dimsAll = Object.keys(recs[0]).filter(k => !["geo", "time", "value", "geoLabel"].includes(k));
  let r = recs;
  for (const d of dimsAll) {
    const cats = [...new Set(r.map(x => x[d]))];
    if (cats.length < 2) continue;
    const elegida = preferidas.find(p => cats.includes(p)) ?? cats[0];
    r = r.filter(x => x[d] === elegida);
  }
  return r;
}

async function eurostat(code, query = "") {
  const url = `https://ec.europa.eu/eurostat/api/dissemination/statistics/1.0/data/${code}?format=JSON&lang=EN${query}`;
  const res = await fetch(url);
  if (!res.ok) {
    let msg = `HTTP ${res.status}`;
    try { const js = await res.json(); const err = Array.isArray(js.error) ? js.error[0] : js.error; if (err && err.label) msg += ` – ${err.label}`; } catch (e) {}
    throw new Error(`Eurostat ${code}: ${msg}`);
  }
  return jsonstat(await res.json());
}

function fecha(t) {
  let m;
  if ((m = /^(\d{4})-(\d{2})$/.exec(t))) return new Date(+m[1], +m[2] - 1, 1);
  if ((m = /^(\d{4})M(\d{2})$/.exec(t))) return new Date(+m[1], +m[2] - 1, 1);
  if ((m = /^(\d{4})-?Q(\d)$/.exec(t))) return new Date(+m[1], (+m[2] - 1) * 3, 1);
  if ((m = /^(\d{4})-?S(\d)$/.exec(t))) return new Date(+m[1], (+m[2] - 1) * 6, 1);
  if ((m = /^(\d{4})$/.exec(t))) return new Date(+m[1], 0, 1);
  return null;
}

async function consultar(code, query, geos, desde) {
  try {
    return await eurostat(code, query + (geos ? geos.map(g => `&geo=${g}`).join("") : "") + desde);
  } catch (e1) {
    // Si algún territorio no existe en la tabla, se repite la consulta sin filtrar territorios
    const r = await eurostat(code, query + desde);
    return geos ? r.filter(x => geos.includes(x.geo)) : r;
  }
}

async function cargar(spec) {
  const desde = `&sinceTimePeriod=${spec.desde ?? 2008}`;
  const geos = spec.regional ? geosRegiones : (spec.sinGeo ? null : geosPaises);
  // Intentos: la consulta principal, las alternativas (p. ej., tablas que Eurostat ha renovado)
  // y, por último, la tabla sin filtros adicionales
  const intentos = [{code: spec.code, query: spec.query}, ...(spec.alt ?? []), {code: spec.code, query: ""}];
  let recs = null, usado = null, error = "";
  for (const t of intentos) {
    try { recs = await consultar(t.code, t.query, geos, desde); if (recs.length) { usado = t; break; } }
    catch (e) { error = e.message; }
  }
  if (!recs || !recs.length) return {ok: false, error: error || "la consulta no devolvió datos", rows: []};
  recs = unaCategoria(recs);
  const rows = recs.map(r => ({
    indicador: spec.nombre,
    codigo_territorio: r.geo ?? "",
    territorio: r.geo ? (nombresGeo[r.geo] ?? r.geoLabel) : (spec.serie ?? spec.nombre),
    periodo: r.time,
    fecha: fecha(r.time),
    valor: r.value,
    unidad: spec.unidad,
    fuente: `Eurostat (${usado.code})`
  })).filter(r => r.fecha && Number.isFinite(r.valor));
  return {ok: rows.length > 0, rows, code: usado.code, error: rows.length ? "" : "la consulta no devolvió datos"};
}

function rewind(f) {
  // Los GeoJSON de GISCO siguen la regla de la mano derecha; d3 necesita el sentido contrario
  const fix = poly => d3.geoArea({type: "Polygon", coordinates: poly}) > 2 * Math.PI ? poly.map(r => r.slice().reverse()) : poly;
  const g = f.geometry;
  if (g?.type === "Polygon") g.coordinates = fix(g.coordinates);
  else if (g?.type === "MultiPolygon") g.coordinates = g.coordinates.map(fix);
  return f;
}

async function geoNuts(nivel) {
  // Límites de países (nivel 0) o regiones NUTS 2 (nivel 2) de Eurostat-GISCO; los países tienen
  // como alternativa Natural Earth. Solo se descargan cuando se pide un mapa y se guardan en memoria.
  window.__cceeGeo = window.__cceeGeo ?? new Map();
  if (!window.__cceeGeo.has(nivel)) window.__cceeGeo.set(nivel, geoDescargar(nivel));
  return window.__cceeGeo.get(nivel);
}

async function geoDescargar(nivel) {
  const url = `https://gisco-services.ec.europa.eu/distribution/v2/nuts/geojson/NUTS_RG_20M_2021_4326_LEVL_${nivel}.geojson`;
  try {
    const r = await fetch(url);
    if (r.ok) { const g = await r.json(); g.features.forEach(f => { rewind(f); f.properties.geo = f.properties.NUTS_ID; }); return g; }
  } catch (e) {}
  if (nivel === 0) {
    try {
      const r = await fetch("https://cdn.jsdelivr.net/gh/nvkelso/natural-earth-vector@master/geojson/ne_110m_admin_0_countries.geojson");
      const g = await r.json(), eq = {GR: "EL", GB: "UK"};
      g.features.forEach(f => { const c = f.properties.ISO_A2_EH ?? f.properties.ISO_A2; f.properties.geo = eq[c] ?? c; });
      return g;
    } catch (e) {}
  }
  return null;
}

function tooltipMapa(node) {
  // Etiqueta propia para los mapas: aparece en cuanto el ratón está sobre cualquier territorio
  // (la de Plot solo se activa cerca del centro del territorio y fallaba en los grandes).
  const tip = window.__cceeTip ?? (window.__cceeTip = Object.assign(document.createElement("div"), {className: "mapa-tip", hidden: true}));
  if (!tip.isConnected) document.body.appendChild(tip);
  // Los contornos destacados y los nombres no deben tapar a los territorios
  node.querySelectorAll('svg g[fill="none"], svg g[aria-label="text"]').forEach(g => { g.style.pointerEvents = "none"; });
  node.querySelectorAll("svg title").forEach(t => {
    const el = t.parentNode; el.dataset.tip = t.textContent; el.classList.add("mapa-zona"); t.remove();
  });
  let actual = null;
  const ocultar = () => { tip.hidden = true; if (actual) actual.classList.remove("mapa-hl"); actual = null; };
  const mover = e => {
    const el = e.target.closest ? e.target.closest("[data-tip]") : null;
    if (!el || !node.contains(el)) return ocultar();
    if (el !== actual) {
      if (actual) actual.classList.remove("mapa-hl");
      actual = el; el.classList.add("mapa-hl");
      const [l1, ...resto] = el.dataset.tip.split("\n");
      tip.innerHTML = `<b>${l1}</b>${resto.length ? "<br>" + resto.join("<br>") : ""}`;
      tip.hidden = false;
    }
    const x = e.clientX + 14, y = e.clientY + 14, w = tip.offsetWidth, h = tip.offsetHeight;
    tip.style.left = Math.min(x, window.innerWidth - w - 8) + "px";
    tip.style.top = (y + h > window.innerHeight - 8 ? e.clientY - h - 10 : y) + "px";
  };
  node.addEventListener("pointermove", mover);
  node.addEventListener("pointerdown", mover);
  node.addEventListener("pointerleave", ocultar);
  window.addEventListener("scroll", ocultar, {passive: true});
  return node;
}

DJ = {
  const paises = geosPaises.filter(c => /^[A-Z]{2}$/.test(c));
  const regiones = geosRegiones.filter(c => /^ES\d\d$/.test(c) && !["ES63", "ES64"].includes(c));
  // Último periodo con dato para la mayor parte de los territorios
  function ultimoComun(rows, validos, cobertura = 0.6) {
    const por = d3.group(rows.filter(r => validos.includes(r.codigo_territorio)), r => r.periodo);
    const min = Math.max(3, Math.ceil(cobertura * validos.length));
    const p = [...por.keys()].sort().reverse().find(p => new Set(por.get(p).map(r => r.codigo_territorio)).size >= min);
    return p ? {periodo: p, valores: new Map(por.get(p).map(r => [r.codigo_territorio, r.valor]))} : null;
  }
  // Carga varios indicadores a la vez; ref: valor de la UE (países) o de España (regiones) en el mismo periodo
  async function cargarVarios(specs, validos, refCode) {
    const res = await Promise.all(specs.map(async s => {
      try {
        const r = await cargar(s);
        if (!r.ok) return null;
        const u = ultimoComun(r.rows, validos);
        if (!u) return null;
        const ref = r.rows.find(x => x.periodo === u.periodo && x.codigo_territorio === refCode)?.valor;
        return {...s, ...u, ref, fuente: r.code};
      } catch (e) { return null; }
    }));
    return res.filter(Boolean);
  }
  const nombre = (c) => nombresGeo[c] ?? c;
  function bandera(c) {
    const iso = c === "EU27_2020" ? "eu" : c === "EL" ? "gr" : c === "UK" ? "gb" : (/^[A-Z]{2}$/.test(c) ? c.toLowerCase() : null);
    if (!iso) return "";
    const img = html`<img class="mem-bandera" alt="" loading="lazy">`;
    img.src = `https://flagcdn.com/w80/${iso}.png`;
    img.onerror = () => img.remove();
    return img;
  }
  function barajar(a, azar = Math.random) { const b = [...a]; for (let i = b.length - 1; i > 0; i--) { const j = Math.floor(azar() * (i + 1)); [b[i], b[j]] = [b[j], b[i]]; } return b; }
  const conUnidad = (v, uc) => uc === "%" ? `${fmt(v)} %` : uc ? `${fmt(v)} ${uc}` : fmt(v);
  return {paises, regiones, ultimoComun, cargarVarios, nombre, bandera, barajar, conUnidad};
}
